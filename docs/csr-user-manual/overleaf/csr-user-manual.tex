\section{Introduction}

\textbf{CSR} stands for \textbf{Customer Support Representative}. In the BEM CRM system, the CSR panel is the daily workspace for handling assigned client calls, managing the sales pipeline, closing deals, and tracking active projects.

\subsection{What You Can Do}
\begin{itemize}[noitemsep]
    \item Work your \textbf{assigned leads} only
    \item Update lead statuses and add notes
    \item Manage pipeline: Call Data $\rightarrow$ Potential Clients $\rightarrow$ Follow-up $\rightarrow$ Clients $\rightarrow$ Active Projects
    \item Add leads manually and detect returning clients by phone
    \item Close clients (won deals) and manage active projects
    \item Chat with team members (DMs, groups, files, voice)
    \item View your personal activity report
    \item Enable/disable Two-Factor Authentication (2FA)
\end{itemize}

\subsection{What You Cannot Do}
\begin{itemize}[noitemsep]
    \item Access Admin or Manager panels
    \item View other CSRs' leads (except when reassigning at retry limits)
    \item Manage users, bulk uploads, or system settings
\end{itemize}

\begin{warningbox}
Login must use the \textbf{CSR portal} (\texttt{/login/csr}). Admin/Manager portals will not work for CSR accounts. Only \textbf{one active session} is allowed at a time.
\end{warningbox}

\section{Login}

\begin{workflowbox}
\begin{enumerate}[noitemsep]
    \item Open the CRM home page
    \item Click \textbf{CSR} portal $\rightarrow$ navigates to \texttt{/login/csr}
    \item Enter \textbf{Email} and \textbf{Password}
    \item Optionally tick \textbf{Remember me}
    \item If prompted, complete \textbf{CAPTCHA} (after failed login attempts)
    \item If 2FA is enabled, enter the \textbf{6-digit OTP} sent to your email
    \item You are redirected to \textbf{Call Data} (\texttt{/csr})
\end{enumerate}
\end{workflowbox}

\section{Lead Status Reference}

Every lead has a \textbf{status} that determines where it appears in the system.

\begin{longtable}{@{}p{4cm}p{10cm}@{}}
\toprule
\textbf{Status} & \textbf{Meaning} \\
\midrule
\endhead
Pending             & Not yet worked / in queue \\
Not Picked          & Phone was not answered \\
Not Interested      & Client declined interest \\
No Owner Available  & Correct contact person not found \\
Not In Service      & Our service is not relevant to client \\
In House            & Internal / special case \\
Important           & Requires follow-up (no schedule yet) \\
Schedule Call       & Follow-up date and time are set \\
Interested          & Client showed interest; service selected \\
Close Client        & Deal won --- project starts \\
\bottomrule
\end{longtable}

\begin{warningbox}
\textbf{Retry Limits:}
\begin{itemize}[noitemsep]
    \item \textbf{Not Interested:} 3 tries $\rightarrow$ Bin or Reassign to another CSR
    \item \textbf{Not Picked:} 7 tries $\rightarrow$ Bin or Reassign to another CSR
\end{itemize}
\end{warningbox}

\section{Common Workflows}

\subsection{Work a New Assigned Lead}

\begin{workflowbox}
\begin{enumerate}
    \item Go to \textbf{Call Data} $\rightarrow$ \textbf{Today Leads} or \textbf{Pending} tab
    \item Apply filters if needed (state, timezone, source)
    \item Click \textbf{eye icon} on client name to review details
    \item Add a \textbf{note} summarizing the call
    \item Select \textbf{Update Status} based on call outcome:
    \begin{itemize}
        \item No answer $\rightarrow$ \textbf{Not Picked}
        \item Declined $\rightarrow$ \textbf{Not Interested}
        \item Follow-up needed $\rightarrow$ \textbf{Important}
        \item Schedule call $\rightarrow$ \textbf{Schedule Call} (set date, time, timezone)
        \item Shows interest $\rightarrow$ \textbf{Interested} (select service)
        \item Deal won $\rightarrow$ \textbf{Close Client}
    \end{itemize}
\end{enumerate}
\end{workflowbox}

\subsection{Schedule a Call}

\begin{workflowbox}
\begin{enumerate}
    \item From Call Data: select \textbf{Schedule Call} status\\
          \textit{OR} from Potential Clients $\rightarrow$ Edit $\rightarrow$ Move to Schedule Call
    \item Enter \textbf{Date}, \textbf{Time}, and \textbf{Timezone} (US timezones)
    \item Optionally add a note
    \item Save $\rightarrow$ lead appears under \textbf{Schedule Call} tab
    \item Notification bell will also show the scheduled call
\end{enumerate}
\end{workflowbox}

\subsection{Mark Client as Interested}

\begin{workflowbox}
\begin{enumerate}
    \item Select status \textbf{Interested}
    \item Choose one or more services: Estimating, Scheduling, Shop Drawing, Drafting, BIM, IT
    \item Optionally select a subtype for each service
    \item Save $\rightarrow$ lead moves to \textbf{Interested Clients} tab on Potential Clients
\end{enumerate}
\end{workflowbox}

\subsection{Close a Client (Deal Won)}

\begin{workflowbox}
\begin{enumerate}
    \item Select \textbf{Close Client} from any page
    \item Fill required fields:
    \begin{itemize}
        \item \textbf{Project Title}
        \item \textbf{Deadline}
        \item \textbf{Service}
        \item \textbf{Paid Amount}
    \end{itemize}
    \item Optional: Project Code, Budget, Notes, Email
    \item Save $\rightarrow$ system auto-generates Client Code (e.g.\ \texttt{CSR001-01})
    \item You are redirected to \textbf{Active Projects}
\end{enumerate}
\end{workflowbox}

\subsection{Retry Not Interested / Not Picked}

\begin{workflowbox}
\begin{enumerate}
    \item Go to \textbf{Not Interested} nav $\rightarrow$ relevant tab
    \item Click \textbf{Retry} button (increments try count)
    \item At limit (NI = 3, NP = 7): \textbf{Assign to CSR} modal opens
    \item Select a peer CSR to reassign, \textit{or} change status back to Pending / Important / Interested
\end{enumerate}
\end{workflowbox}

\subsection{Restore from Bin}

\begin{workflowbox}
\begin{enumerate}
    \item Go to \textbf{Bin} page
    \item Find the lead (use search or tabs)
    \item Click \textbf{Restore}
    \item Lead returns to normal workflow
\end{enumerate}
\end{workflowbox}

\section{Tips \& Rules}

\begin{tipbox}
\begin{itemize}
    \item You can only see \textbf{your assigned leads}
    \item Always \textbf{add notes} after each call --- history is valuable
    \item Watch \textbf{retry limits}: Not Interested = 3, Not Picked = 7
    \item Check the \textbf{notification bell} for scheduled calls
    \item Entering a phone number may \textbf{auto-detect returning clients}
    \item Client codes are auto-generated on Close Client (\texttt{CSR001-01} format)
    \item Always \textbf{logout} at end of shift via Profile Menu
    \item Use \textbf{Messages} to coordinate with your team
\end{itemize}
\end{tipbox}
